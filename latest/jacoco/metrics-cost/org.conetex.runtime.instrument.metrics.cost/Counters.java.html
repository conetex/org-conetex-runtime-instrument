    public static void reset() {
        for(Counter c : COUNTERS){
            c.reset();
        }
    }

    public static void blockIncrement(boolean incrementationBlocked) {
        for(Counter c : COUNTERS){
            c.blockIncrement(incrementationBlocked);
        }
    }
